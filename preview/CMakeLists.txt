cmake_minimum_required(VERSION 3.16)
project(pet_preview LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(PET_LVGL_DIR "${CMAKE_CURRENT_LIST_DIR}/../managed_components/lvgl__lvgl" CACHE PATH "LVGL 9.5 source")
add_subdirectory("${PET_LVGL_DIR}" lvgl)
string(TIMESTAMP PET_BUILD_EPOCH "%s" UTC)
configure_file(../main/pet_clock_build.h.in "${CMAKE_CURRENT_BINARY_DIR}/pet_clock_build.h" @ONLY)
add_executable(pet_preview render.c store_stub.c device_stub.c audio_stub.c ../main/pet_sound.c ../main/pet_model.c ../main/pet_routine.c ../main/pet_ui.c ../main/pet_font.c ../main/pet_save.c ../main/pet_battery.c ../main/pet_touch.c)
target_include_directories(pet_preview PRIVATE ../main "${CMAKE_CURRENT_BINARY_DIR}")
# The renderer also verifies touch/state behavior; keep assertions in Release.
target_compile_options(pet_preview PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(pet_preview PRIVATE lvgl m)
